from PIL import Image, ImageDraw, ImageFont import os def crear_plantilla_fondo(output_path="plantilla_base.jpg"): # Dimensiones estándar del lienzo W, H = 1100, 1550 out = Image.new("RGB", (W, H), "white") d = ImageDraw.Draw(out) # Cargar fuentes (con respaldo por defecto si no están disponibles) try: font_titulo = ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", 32) font_texto = ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", 18) except OSError: font_titulo = ImageFont.load_default() font_texto = ImageFont.load_default() # 1. Cabecera (Seudónimo y Fecha) d.text((W // 2, 45), "[ SEUDÓNIMO ]", font=font_titulo, fill="#999999", anchor="mm") d.text((W // 2, 95), "Fecha: DD/MM/AAAA", font=font_texto, fill="#999999", anchor="mm") # 2. Marco contenedor para la fotografía corporal area = (75, 180, 780, 1430) d.rectangle(area, outline="#d0d0d0", width=2) d.text((area[0] + 30, area[1] + 30), "Área de fotografía corporal", font=font_texto, fill="#b0b0b0") # 3. Regla vertical derecha (Escala referencial) ybottom = 1430 ytop = 210 cmmax = 190 for cm in range(0, cmmax + 1, 5): y = round(ybottom - (cm / cmmax) * (ybottom - ytop)) d.line((790, y, 810 + (12 if cm % 10 == 0 else 0), y), fill="#888888", width=1) if cm % 10 == 0: d.text((835, y - 8), str(cm), font=font_texto, fill="#666666") # 4. Cuadrícula en perspectiva para la zona de los pies floor_top = 1300 floor_bottom = 1430 # Líneas de profundidad horizontales for i in range(6): y = floor_top + i * 26 d.line((250 + i * 12, y, 780 - i * 12, y), fill="#e0e0e0", width=1) # Líneas convergentes en perspectiva for x in range(250, 781, 50): d.line((x, floor_top, 515 + (x - 515) * 0.15, floor_bottom), fill="#e0e0e0", width=1) # Guardar imagen resultante out.save(output_path, "JPEG", quality=95) print(f"Plantilla generada y guardada correctamente en: {output_path}") if __name__ == "__main__": crear_plantilla_fondo()
